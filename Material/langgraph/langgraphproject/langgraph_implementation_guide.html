3 Define State Schema

What We'll Do
Define typed state schemas for the main graph and subgraphs. This is the foundation of LangGraph - state flows through all nodes and determines how data is shared between agents.
🎓 Key Concepts from LangGraph Basics
  • State Schema: Use TypedDict or Pydantic models to define what data flows through your graph
  • MessagesState: LangGraph provides MessagesState with built-in message history
  • Reducers: Use Annotated[list, operator.add] for fields that need to accumulate results from parallel nodes
  • Subgraph State: Each subgraph has its own state schema - only input/output fields need to be in main state
✏️ Your Task

Create src/ecommerce_support/state/schemas.py:

"""State schemas for the e-commerce support system."""
from typing import Annotated, Literal, TypedDict, Optional
from typing_extensions import TypedDict as ExtTypedDict
from langgraph.graph import MessagesState
import operator
from pydantic import BaseModel, Field


class AgentState(MessagesState):
    """
    Main graph state schema extending MessagesState.
    
    This state flows through the supervisor and all worker agents.
    It contains:
    - User context and conversation tracking
    - Routing decisions for supervisor
    - Inputs to pass to subgraphs
    - Outputs received from subgraphs
    - Parallel processing results
    """
    
    # User context
    user_email: str
    conversation_id: str
    
    # Routing decisions (supervisor uses these)
    next_agent: Optional[Literal["order_agent", "inventory_agent", "policy_agent", "FINISH"]] = None
    
    # Subgraph inputs (main graph → subgraph)
    order_id: Optional[int] = None
    return_items: Optional[list[dict]] = None  # [{"product_id": 1, "quantity": 1}]
    return_reason: Optional[str] = None
    
    # Subgraph outputs (subgraph → main graph)
    return_id: Optional[int] = None
    ticket_id: Optional[int] = None
    fraud_score: Optional[float] = None
    refund_amount: Optional[float] = None
    
    # Parallel processing results (uses operator.add reducer to combine from multiple nodes)
    parallel_results: Annotated[list[str], operator.add] = Field(default_factory=list)
    
    # Workflow metadata
    current_workflow: Optional[Literal["cancellation", "return", "general_query"]] = None
    requires_hitl: bool = False


class ReturnsSubgraphState(TypedDict):
    """
    State schema for the Returns Processing Subgraph.
    
    This is a self-contained state for the return workflow that includes
    human-in-the-loop (HITL) approval process.
    """
    
    # Input from main graph
    user_email: str
    conversation_id: str
    order_id: str  # TEXT like 'ORD001'
    return_items: list[dict]  # [{"product_id": str (like 'SKU001'), "quantity": int}]
    return_reason: str
    
    # Internal processing state
    validation_result: Optional[dict] = None  # {"eligible": bool, "message": str}
    return_id: Optional[int] = None
    ticket_id: Optional[int] = None
    
    # Parallel task results (fraud checks, uses operator.add reducer)
    fraud_checks: Annotated[list[dict], operator.add] = Field(default_factory=list)
    fraud_score: Optional[float] = None
    
    # Email and refund
    refund_amount: Optional[float] = None
    email_sent: bool = False
    
    # HITL state (admin decision)
    admin_decision: Optional[Literal["approved", "rejected", "more_info"]] = None
    admin_notes: Optional[str] = None
    
    # Final status
    status: Literal["pending", "processing", "approved", "rejected", "completed"] = "pending"
    
    # Output to main graph (these fields get passed back)
    output: Optional[dict] = None  # Contains: return_id, ticket_id, fraud_score, refund_amount, status


class FraudDetectionState(TypedDict):
    """
    State schema for the Fraud Detection Subgraph.
    
    This subgraph runs parallel checks on return requests.
    """
    
    # Input
    user_email: str
    order_id: int
    return_items: list[dict]
    return_reason: str
    
    # Parallel check results (uses operator.add reducer)
    check_results: Annotated[list[dict], operator.add] = Field(default_factory=list)
    
    # Aggregated output
    fraud_score: float = 0.0

def load_policy_documents():
    """
    Load all policy documents from the policies directory.
    
    Returns:
        List of Document objects
    """
    policies_dir = Config.POLICIES_DIR
    
    # Load markdown files
    loader = DirectoryLoader(
        str(policies_dir),
        glob="**/*.md",

    )
    
    documents = loader.load()

    return documents


def create_vector_store(documents: list[Document] = None, force_recreate: bool = False):
    """
    Create or load ChromaDB vector store with policy documents.
    
    Args:
        documents: List of documents to add. If None, loads from policies directory.
        force_recreate: If True, deletes existing store and creates new one.
        
    Returns:
        Chroma vector store instance
    """
    chroma_path = Path(Config.CHROMA_DB_PATH)
    
    # Delete existing store if force_recreate
    if force_recreate and chroma_path.exists():
        import shutil
        shutil.rmtree(chroma_path)
        print("🗑️  Deleted existing vector store")
    
    # Create embeddings
    embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
    
    # If vector store exists and not force_recreate, load it
    if chroma_path.exists() and not force_recreate:
        print("📂 Loading existing vector store...")
        vectorstore = Chroma(
            persist_directory=str(chroma_path),
            embedding_function=embeddings
        )
        print(f"✅ Loaded vector store with {vectorstore._collection.count()} documents")
        return vectorstore
    
    # Create new vector store
    if documents is None:
        documents = load_policy_documents()
    
    # Split documents into chunks
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=200,
        length_function=len,
        separators=["\n\n", "\n", " ", ""]
    )
    
    splits = text_splitter.split_documents(documents)
    print(f"✅ Split into {len(splits)} chunks")
    
    # Create vector store
    vectorstore = Chroma.from_documents(
        documents=splits,
        embedding=embeddings,
        persist_directory=str(chroma_path)
    )
    
    print(f"✅ Created vector store at {chroma_path}")
    print(f"✅ Stored {vectorstore._collection.count()} document chunks")
    
    return vectorstore


def get_policy_retriever(k: int = 4):
    """
    Get a retriever for policy documents.
    
    Args:
        k: Number of documents to retrieve
        
    Returns:
        Retriever instance that can be used as a tool
    """
    vectorstore = create_vector_store()
    retriever = vectorstore.as_retriever(
        search_type="similarity",
        search_kwargs={"k": k}
    )
    return retriever


def search_policies(query: str, k: int = 4):
    """
    Search policy documents for relevant information.
    
    Args:
        query: Search query
        k: Number of results to return
        
    Returns:
        List of relevant document chunks
    """
    retriever = get_policy_retriever(k=k)
    results = retriever.invoke(query)
    return results

SCOPES = ['https://www.googleapis.com/auth/gmail.send']
_gmail_service = None
_gmail_initialized = False

def get_gmail_service():
    """Get Gmail API service with OAuth2 authentication."""
    global _gmail_service, _gmail_initialized
    
    if _gmail_initialized:
        return _gmail_service
    
    credentials_path = Path("credentials.json")
    token_path = Path("token.pickle")
    
    if not credentials_path.exists():
        print("⚠️  Gmail credentials not found: credentials.json")
        print("   Gmail functionality disabled. See setup instructions.")
        _gmail_initialized = True
        _gmail_service = None
        return None
    
    creds = None
    if token_path.exists():
        with open(token_path, 'rb') as token:
            creds = pickle.load(token)
    
    if not creds or not creds.valid:
        if creds and creds.expired and creds.refresh_token:
            creds.refresh(Request())
        else:
            flow = InstalledAppFlow.from_client_secrets_file(
                str(credentials_path), SCOPES
            )
            creds = flow.run_local_server(port=0)
        
        with open(token_path, 'wb') as token:
            pickle.dump(creds, token)
    
    _gmail_service = build('gmail', 'v1', credentials=creds)
    _gmail_initialized = True
    print("✅ Gmail API authenticated")
    return _gmail_service

def get_gmail_tools():
    """Get Gmail tools for agents."""
    service = get_gmail_service()
    return [GmailSendMessage(api_resource=service)] if service else []

def is_gmail_available():
    """Check if Gmail is configured."""
    return get_gmail_service() is not None

"""Supervisor agent for routing queries to worker agents."""
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
from langchain_openai import ChatOpenAI
from pydantic import BaseModel, Field
from ecommerce_support.config import Config


class RouteDecision(BaseModel):
    """Structured output for supervisor routing decisions."""
    
    next: str = Field(
        description="The name of the next agent to route to, or FINISH if complete"
    )
    reasoning: str = Field(
        description="Brief explanation of why this agent was chosen"
    )


def create_supervisor_agent(worker_names: list[str], llm: ChatOpenAI = None):
    """
    Create a supervisor agent that routes to worker agents.
    
    The supervisor uses an LLM with structured output to analyze queries
    and decide which worker agent should handle them, or when to finish.
    
    Args:
        worker_names: List of available worker agent names
        llm: ChatOpenAI instance. If None, uses SUPERVISOR_MODEL from config.
        
    Returns:
        Supervisor function that takes state and returns routing decision
    """
    if llm is None:
        llm = ChatOpenAI(model=Config.SUPERVISOR_MODEL, temperature=0)
    
    # Create static system prompt
    worker_descriptions = {
        "OrderAgent": (
            "Handles order status, payment information, order cancellations. "
            "Can query orders, order_items, and payments tables."
        ),
        "InventoryAgent": (
            "Checks product availability, stock levels, product details, and pricing. "
            "Can query products table."
        ),
        "PolicyAgent": (
            "Answers questions about policies (returns, shipping, FAQ) using RAG. "
            "Does not access database - uses policy documents."
        )
    }
    
    # Build the system prompt
    workers_info = "\n".join([
        f"- {name}: {worker_descriptions.get(name, 'Worker agent')}"
        for name in worker_names
    ])
    
    system_prompt = f"""You are a supervisor managing an e-commerce customer support system.

Your team of workers:
{workers_info}

Your responsibilities:
1. Analyze customer queries to determine which worker agent is most appropriate
2. Route queries to the correct worker agent
3. Review worker responses and determine if additional workers are needed
4. Respond with FINISH when the customer's query has been fully addressed

Routing guidelines:
- Order-related queries (status, payments, cancellations, returns) → OrderAgent
- Return/refund requests → OrderAgent (validates and invokes returns subgraph with HITL)
- Product/inventory queries (availability, stock, pricing) → InventoryAgent  
- Policy questions (returns, shipping, FAQ) → PolicyAgent
- Multi-part queries may require multiple agents sequentially
- Return FINISH only when all parts of the query are fully answered

Important routing logic:
- Consider the FULL conversation history when making routing decisions
- If a worker has already handled their part completely, move to the next needed worker or FINISH
- **CRITICAL: When an agent asks the user for information (missing order ID, return reason, etc.):**
  - Return FINISH immediately to let the user respond
  - When the user provides the requested information in their next message, route BACK to the SAME agent that asked the question
  - Example: OrderAgent asks "What's your order ID?" → FINISH → User says "ORD001" → Route to OrderAgent
- **Pattern recognition for follow-ups:**
  - If last agent message contains questions like "What is...", "Which order...", "Can you provide..." AND the new user message appears to answer that question → Route to the agent that asked
  - If user message references something the previous agent was handling (order ID, product name, return request) → Route to that same agent
- Don't route to the same agent multiple times in a row UNLESS the user is providing requested information
- Be efficient - finish as soon as the query is fully addressed
"""
    
    options = ["FINISH"] + worker_names
    
    # Create the prompt with message history
    prompt = ChatPromptTemplate.from_messages([
        ("system", system_prompt),
        MessagesPlaceholder(variable_name="messages"),
        ("system", 
         f"Given the conversation above, who should act next? "
         f"Or should we FINISH? Select one of: {options}\n\n"
         f"Provide your reasoning for the decision.")
    ])
    
    # Create the supervisor chain with structured output
    supervisor_chain = prompt | llm.with_structured_output(RouteDecision)
    
    def supervisor_node(state):
        """Supervisor node function for LangGraph."""
        result = supervisor_chain.invoke(state)
        print(f"\n🧭 Supervisor Decision: {result.next}")
        print(f"   Reasoning: {result.reasoning}")
        return {"next": result.next}
    
    return supervisor_node

Part 2: Checkpointer Setup Function

This function manages the PostgreSQL checkpointer for conversation persistence. It handles the context manager properly to avoid connection issues:

def get_checkpointer() -> PostgresSaver:
    """
    Get or create the shared PostgresSaver checkpointer instance.
    
    Requires POSTGRES_CONNECTION_STRING environment variable.
    Format: postgresql://user:password@host:port/database
    
    Note: PostgresSaver.from_conn_string() returns a context manager.
    We enter it once and keep both the context manager and checkpointer
    alive for the application lifetime to prevent connection from closing.
    
    Returns:
        PostgresSaver instance for conversation persistence
        
    Raises:
        ValueError: If POSTGRES_CONNECTION_STRING is not set
    """
    global _checkpointer, _context_manager
    if _checkpointer is None:
        conn_string = os.getenv("POSTGRES_CONNECTION_STRING")
        if not conn_string:
            raise ValueError(
                "POSTGRES_CONNECTION_STRING environment variable is required. "
                "Format: postgresql://user:password@host:port/database"
            )
        # from_conn_string returns a context manager
        # Store both the context manager and checkpointer to keep connection alive
        _context_manager = PostgresSaver.from_conn_string(conn_string)
        _checkpointer = _context_manager.__enter__()
        # Setup database tables (safe to call multiple times)
        _checkpointer.setup()
    return _checkpointer

Why this matters: PostgresSaver.from_conn_string() returns a context manager, not the checkpointer directly. We use __enter__() to get the actual checkpointer and keep both alive to maintain the database connection.

Part 3: Agent Node Factory Function

This function creates node functions for worker agents using the closure pattern:

def create_agent_node(agent, name: str):
    """
    Create a node function for an agent using closure pattern.
    
    This factory function creates a node that:
    1. Invokes the agent with the current state
    2. Extracts the agent's response
    3. Wraps it in a HumanMessage with the agent's name
    4. Returns it to be added to state
    
    Args:
        agent: The agent (created with create_agent)
        name: The name to use for this agent's messages
        
    Returns:
        Node function for use in LangGraph
    """
    def node(state: AgentState):
        result = agent.invoke(state)
        message_content = result["messages"][-1].content
        
        print(f"\n💬 {name}: {message_content[:150]}...")
        
        return {
            "messages": [HumanMessage(content=message_content, name=name)]
        }
    
    return node

How it works: The outer function captures agent and name, and returns an inner function that LangGraph can call as a node. This is called the "closure pattern" - the inner function "closes over" the variables from the outer function.

def create_support_graph(checkpointer=None):
    """
    Create the complete e-commerce support graph with all agents.
    
    Graph structure:
    START → supervisor → [OrderAgent, InventoryAgent, PolicyAgent] → supervisor → ... → END
    
    The supervisor routes to worker agents based on the query, and workers
    return to the supervisor. This continues until supervisor returns FINISH.
    
    Args:
        checkpointer: Optional checkpointer for conversation persistence (e.g., PostgresSaver)
                     If None, defaults to MemorySaver for in-memory persistence
        
    Returns:
        Compiled LangGraph application
    """
    # Initialize checkpointer with MemorySaver if not provided
    if checkpointer is None:
        checkpointer = MemorySaver()
    
    # Create LLM instances
    worker_llm = ChatOpenAI(model=Config.DEFAULT_MODEL, temperature=0)
    supervisor_llm = ChatOpenAI(model=Config.SUPERVISOR_MODEL, temperature=0)
    
    # Create worker agents
    print("🔨 Creating worker agents...")
    order_agent = create_order_agent(worker_llm)
    inventory_agent = create_inventory_agent(worker_llm)
    policy_agent = create_policy_agent(worker_llm)
    
    # Create agent nodes
    order_node = create_agent_node(order_agent, "OrderAgent")
    inventory_node = create_agent_node(inventory_agent, "InventoryAgent")
    policy_node = create_agent_node(policy_agent, "PolicyAgent")
    
    # Create supervisor
    print("🧭 Creating supervisor agent...")
    worker_names = ["OrderAgent", "InventoryAgent", "PolicyAgent"]
    supervisor = create_supervisor_agent(worker_names, supervisor_llm)
    
    # Create the graph
    print("📊 Building LangGraph workflow...")
    workflow = StateGraph(AgentState)
    
    # Add nodes
    workflow.add_node("supervisor", supervisor)
    workflow.add_node("OrderAgent", order_node)
    workflow.add_node("InventoryAgent", inventory_node)
    workflow.add_node("PolicyAgent", policy_node)
    
    # Add edges - all workers return to supervisor
    workflow.add_edge("OrderAgent", "supervisor")
    workflow.add_edge("InventoryAgent", "supervisor")
    workflow.add_edge("PolicyAgent", "supervisor")
    
    # Start with supervisor
    workflow.add_edge(START, "supervisor")
    
    # Conditional edges from supervisor
    workflow.add_conditional_edges(
        "supervisor",
        lambda x: x["next"],  # Get routing decision from state
        {
            "OrderAgent": "OrderAgent",
            "InventoryAgent": "InventoryAgent",
            "PolicyAgent": "PolicyAgent",
            "FINISH": END
        }
    )
    
    # Compile the graph
    print("⚙️ Compiling graph...")
    graph = workflow.compile(checkpointer=checkpointer)
    
    print("✅ Support graph created successfully!")
    return graph

Graph Flow:

  • START → supervisor: All queries begin at the supervisor
  • supervisor → workers: Supervisor routes to OrderAgent, InventoryAgent, or PolicyAgent based on query analysis
  • workers → supervisor: All workers return to supervisor after completing their task
  • supervisor → FINISH: When query is fully answered, supervisor routes to END

Step 8: Returns Subgraph with Human-in-the-Loop (HITL)

🎯 What You'll Build in This Step
You'll create a returns processing subgraph that demonstrates the Human-in-the-Loop (HITL) pattern using LangGraph interrupts. This is one of the most powerful features of LangGraph, allowing you to pause execution for human approval before proceeding with critical actions.

Key Concepts:
• Subgraphs: Self-contained workflows with their own state and logic
• Interrupts: Pause execution at any point and wait for external input
• HITL Pattern: Human approval before critical actions (returns, refunds)
• Checkpointers: Required for interrupts to save/restore state
• Conditional Routing: Different paths based on approval/rejection

Workflow:
1. Customer requests return → Validate eligibility → Create return record
2. ⏸️ Interrupt for admin approval (graph pauses here)
3. Admin reviews request → Approves or rejects
4. Graph resumes → Process approval (refund) OR rejection (notify customer)
⚠️ Critical Understanding
Interrupts are NOT like function calls! When interrupt() is called:

1. The entire graph execution stops immediately
2. State is saved to the checkpointer
3. Control returns to the caller with special __interrupt__ marker
4. Graph waits indefinitely until you call graph.invoke(Command(resume=value))
5. When resumed, the entire node restarts from the beginning
6. The value passed to Command(resume=...) becomes the return value of interrupt()

This means:
• ✅ Place side effects (database writes, API calls) AFTER interrupts
• ✅ Use same thread_id when resuming
• ✅ Always use a checkpointer (MemorySaver, PostgresSaver, etc.)
• ❌ Don't wrap interrupt() in try/except
• ❌ Don't conditionally skip interrupts (they must execute in consistent order)

8.1 Understanding the Returns Workflow Architecture

The returns subgraph is a self-contained workflow that can be invoked by the Order Agent when it detects a return request. Unlike the supervisor routing pattern, subgraphs are not part of the main routing logic—they're specialized workflows that handle complex multi-step processes.

📐 Architecture Pattern: Subgraph vs. Worker Agent

Worker Agent (Order/Inventory/Policy):
• Single-step tasks
• No internal state management
• Routed by supervisor
• Example: "What's my order status?" → Query DB → Return answer

Subgraph (Returns Processing):
• Multi-step workflow
• Own state schema and internal routing
• Invoked by agents (not routed by supervisor)
• Can have interrupts (HITL)
• Example: Return request → Validate → Create record → ⏸️ Approval → Process → Send email

When to use subgraphs:
• Complex workflows with multiple decision points
• Need for human approval (HITL)
• Internal state that shouldn't pollute main graph
• Reusable across different agents

def request_approval_node(state: ReturnsSubgraphState) -> dict:
    """
    Request human approval for the return using LangGraph interrupt.
    
    ⏸️ THIS IS WHERE THE HITL PATTERN HAPPENS!
    
    When interrupt() is called:
    1. Graph execution stops immediately
    2. State is saved to checkpointer
    3. Control returns to caller with __interrupt__ marker
    4. Caller checks the interrupt payload
    5. Admin reviews and provides decision
    6. Caller resumes with graph.invoke(Command(resume=decision))
    7. Node restarts from beginning, interrupt() returns the decision
    """
    print("⏸️  Requesting admin approval (HITL)...")
    
    if not state["validation_result"]["eligible"]:
        return {}  # Skip if validation failed
    
    # Prepare approval request data
    approval_request = {
        "type": "return_approval_request",
        "return_id": state["return_id"],
        "ticket_id": state["ticket_id"],
        "order_id": state["order_id"],
        "user_email": state["user_email"],
        "return_reason": state["return_reason"],
        "refund_amount": state["refund_amount"],
        "items": state["validation_result"].get("items", []),
        "days_since_delivery": state["validation_result"].get("days_since_order", 0),
        "question": "Do you approve this return request?",
        "instructions": "Respond with: {'decision': 'approved', 'notes': '...'} or {'decision': 'rejected', 'notes': '...'}"
    }
    
    print(f"\n📋 Return Approval Request:")
    print(f"   Return ID: {approval_request['return_id']}")
    print(f"   Order ID: {approval_request['order_id']}")
    print(f"   Customer: {approval_request['user_email']}")
    refund_amt = approval_request['refund_amount']
    print(f"   Refund: ${refund_amt:.2f}" if refund_amt is not None else "   Refund: Not calculated")
    print(f"   Reason: {approval_request['return_reason']}")
    print(f"\n⏸️  Execution paused. Waiting for admin decision...")
    
    # 🚨 INTERRUPT HERE - execution pauses until admin provides decision
    admin_response = interrupt(approval_request)
    
    print(f"\n▶️  Resumed! Admin decision received: {admin_response}")
    
    # Parse admin response
    if isinstance(admin_response, dict):
        decision = admin_response.get("decision", "rejected")
        notes = admin_response.get("notes", "No notes provided")
    elif isinstance(admin_response, str):
        decision = "approved" if "approve" in admin_response.lower() else "rejected"
        notes = admin_response
    else:
        decision = "rejected"
        notes = "Invalid response format"
    
    return {"admin_decision": decision, "admin_notes": notes}

The HITL magic: The interrupt() call pauses the entire graph. The state is saved, and control returns to the caller with the approval_request data. An admin can review this data in a UI, make a decision, and resume the graph with their response.

def create_returns_subgraph(checkpointer=None):
    """
    Create the Returns Subgraph with Human-in-the-Loop approval.
    
    Args:
        checkpointer: LangGraph checkpointer (REQUIRED for interrupts!)
                     Use MemorySaver() for testing, PostgresSaver for production.
    
    Returns:
        Compiled returns subgraph
    """
    print("🔨 Creating Returns Subgraph with HITL...")
    
    if checkpointer is None:
        print("⚠️  Warning: No checkpointer provided. Using MemorySaver().")
        checkpointer = MemorySaver()
    
    workflow = StateGraph(ReturnsSubgraphState)
    
    # Add nodes
    workflow.add_node("validate_return", validate_return_node)
    workflow.add_node("create_return_record", create_return_record_node)
    workflow.add_node("request_approval", request_approval_node)
    workflow.add_node("process_approved", process_approved_return_node)
    workflow.add_node("process_rejected", process_rejected_return_node)
    
    # Set entry point and edges
    workflow.set_entry_point("validate_return")
    workflow.add_edge("validate_return", "create_return_record")
    workflow.add_edge("create_return_record", "request_approval")
    
    # Conditional routing after approval
    workflow.add_conditional_edges(
        "request_approval",
        route_after_approval,
        {
            "process_approved": "process_approved",
            "process_rejected": "process_rejected",
            "end": END
        }
    )
    
    workflow.add_edge("process_approved", END)
    workflow.add_edge("process_rejected", END)
    
    # Compile with checkpointer (REQUIRED!)
    graph = workflow.compile(checkpointer=checkpointer)
    
    print("✅ Returns Subgraph created successfully!")
    return graph

💡 Understanding What Just Happened

First Invoke (before interrupt):
• Graph runs: validate → create_record → request_approval
• At interrupt(): execution STOPS, state saved, returns with __interrupt__
• No further nodes execute

After Interrupt (waiting state):
• Graph is "paused" with saved state in checkpointer
• You can check result["__interrupt__"] to see approval request data
• Admin reviews the request in a UI (simulated here with hardcoded decision)

Second Invoke (with Command(resume=...)):
• Graph restores state from checkpointer
• Node restarts from beginning (you'll see validation again)
• When it reaches interrupt(), instead of stopping, interrupt() returns the value from resume
• Execution continues: admin_decision is populated → routing → process_approved → email → END

Why does node restart?
• LangGraph cannot resume in the middle of a function
• It re-runs the entire node, but checkpointer ensures interrupt() returns immediately on resume
• This is why side effects (DB writes, emails) should come AFTER interrupts

8.8 Key Learnings and Best Practices

✅ Best Practices for HITL with Interrupts

1. Checkpointer is REQUIRED
• Use MemorySaver() for testing
• Use PostgresSaver() for production
• Without checkpointer, interrupts will fail

2. Side Effects After Interrupts
• ✅ Place DB writes, API calls, emails AFTER interrupt()
• ✅ Read-only operations (queries) before interrupt are OK
• ❌ Don't create records before interrupt (they'll duplicate on resume)

3. Thread ID Consistency
• MUST use same thread_id when resuming
• Each thread_id has independent state
• Different thread_id = new execution, not resume

4. Interrupt Order Matters
• Multiple interrupts in a node must execute in consistent order
• Don't conditionally skip interrupts
• LangGraph tracks which interrupt you're on

5. Pass JSON-Serializable Values
• Interrupt payload must be JSON-serializable (dicts, lists, strings, numbers)
• Don't pass functions or class instances

6. Error Handling
• Don't wrap interrupt() in try/except (breaks the mechanism)
• Handle errors before or after interrupt, not around it

            messages=[HumanMessage(content=f"Process return for order {order_id}. Reason: {return_reason}")],
            order_id=order_id,
            return_reason=return_reason,
            customer_email=customer_email,

def get_db_connection():

def authenticate_user(email: str, password: str) -> Optional[Dict[str, Any]]:
    """
    Authenticate user with email and password.
    
    Args:
        email: User email address
        password: User password
        
    Returns:
        User dict if authentication successful, None otherwise
    """
    conn = get_db_connection()
    cursor = conn.cursor()
    
    cursor.execute(
        "SELECT email, full_name, role, created_at FROM users WHERE email = ? AND password = ?",
        (email, password)
    )
    
    result = cursor.fetchone()
    conn.close()
    
    if result:
        return {
            'email': result[0],
            'full_name': result[1],
            'role': result[2],
            'created_at': result[3]
        }
    return None


def login(email: str, password: str) -> bool:
    """
    Login user and create Streamlit session.
    
    Args:
        email: User email
        password: User password
        
    Returns:
        True if login successful, False otherwise
    """
    user = authenticate_user(email, password)
    
    if user:
        st.session_state['authenticated'] = True
        st.session_state['user'] = user
        return True
    
    return False


def logout():
    """Logout user and clear session."""
    for key in ['authenticated', 'user', 'messages', 'thread_id', 'current_agent']:
        if key in st.session_state:
            del st.session_state[key]


def is_authenticated() -> bool:
    """Check if user is authenticated."""
    return st.session_state.get('authenticated', False)


def get_current_user() -> Optional[Dict[str, Any]]:
    """Get current authenticated user."""
    if is_authenticated():
        return st.session_state.get('user')
    return None


def check_role(required_role: str) -> bool:
    """
    Check if current user has required role.
    
    Args:
        required_role: Required role ('customer' or 'admin')
        
    Returns:
        True if user has required role, False otherwise
    """
    user = get_current_user()
    if user:
        return user['role'] == required_role
    return False

def check_return_frequency(state: FraudDetectionState) -> dict:
    """Check 1: Return frequency analysis (35% weight).
    
    Analyzes how many returns user has made in last 30 days.
    High frequency = potential serial returner.
    """
    user_email = state["user_email"]
    thirty_days_ago = (datetime.now() - timedelta(days=30)).strftime('%Y-%m-%d %H:%M:%S')
    
    db = get_db_connection()
    
    query = """
        SELECT COUNT(*) as return_count
        FROM returns
        WHERE user_email = ?
        AND return_date >= ?
        AND status != 'REJECTED'
    """
    result = db.execute(query, (user_email, thirty_days_ago)).fetchone()
    return_count = result['return_count'] if result else 0
    
    # Calculate risk score
    if return_count > 5:
        risk_score = min(100, 80 + (return_count - 5) * 4)  # HIGH risk
        risk_reason = f"HIGH risk: {return_count} returns in last 30 days"
    elif return_count >= 3:
        risk_score = 40 + (return_count - 2) * 10  # MEDIUM risk
        risk_reason = f"MEDIUM risk: {return_count} returns in last 30 days"
    else:
        risk_score = return_count * 10  # LOW risk
        risk_reason = f"LOW risk: {return_count} returns in last 30 days"
    
    check_result: FraudCheckResult = {
        "check_type": "frequency",
        "score": risk_score,
        "reason": risk_reason,
        "details": {"return_count": return_count, "period_days": 30}
    }
    
    return {"fraud_checks": [check_result]}


def check_order_value_patterns(state: FraudDetectionState) -> dict:
    """Check 2: Order value pattern analysis (35% weight).
    
    Analyzes return value and user's history of high-value returns.
    High-value items = higher fraud risk.
    """
    user_email = state["user_email"]
    return_items = state["return_items"]
    
    db = get_db_connection()
    
    # Calculate current return value
    return_value = 0.0
    for item in return_items:
        # Get unit price from order_items
        query = """
            SELECT unit_price
            FROM order_items
            WHERE order_id = ? AND product_id = ?
        """
        result = db.execute(query, (state["order_id"], item["product_id"])).fetchone()
        if result:
            return_value += result['unit_price'] * item["quantity"]
    
    # Check user's history of high-value returns
    query = """
        SELECT COUNT(*) as high_value_count
        FROM returns
        WHERE user_email = ?
        AND refund_amount > 500
    """
    result = db.execute(query, (user_email,)).fetchone()
    high_value_history = result['high_value_count'] if result else 0
    
    # Calculate risk score
    if return_value > 1000:
        risk_score = min(100, 70 + (return_value - 1000) / 100)
        risk_reason = f"HIGH risk: ${return_value:.2f} return value"
    elif return_value > 500:
        risk_score = 40 + (return_value - 500) / 20
        risk_reason = f"MEDIUM risk: ${return_value:.2f} return value"
    else:
        risk_score = return_value / 50  # LOW risk
        risk_reason = f"LOW risk: ${return_value:.2f} return value"
    
    # Bonus penalty for high-value history
    if high_value_history > 2:
        risk_score = min(100, risk_score + 20)
        risk_reason += f", {high_value_history} high-value returns in history"
    
    check_result: FraudCheckResult = {
        "check_type": "value",
        "score": risk_score,
        "reason": risk_reason,
        "details": {"return_value": return_value, "high_value_history": high_value_history}
    }
    
    return {"fraud_checks": [check_result]}


def check_return_reason_patterns(state: FraudDetectionState) -> dict:
    """Check 3: Return reason pattern analysis (15% weight).
    
    Analyzes return reason for suspicious patterns:
    - Vague reasons ("changed mind", "don't need")
    - Short reasons (<10 characters)
    - Repeated patterns across user's returns
    """
    user_email = state["user_email"]
    return_reason = state["return_reason"].lower()
    
    db = get_db_connection()
    
    risk_score = 0
    risk_factors = []
    
    # Check for vague/suspicious keywords
    suspicious_keywords = ["changed mind", "don't need", "don't want", "no reason", "mistake"]
    for keyword in suspicious_keywords:
        if keyword in return_reason:
            risk_score += 30
            risk_factors.append(f"Vague reason: '{keyword}'")
            break
    
    # Check for very short reasons
    if len(return_reason) < 10:
        risk_score += 25
        risk_factors.append(f"Very short reason ({len(return_reason)} chars)")
    
    # Check if user has used same reason multiple times
    query = """
        SELECT reason, COUNT(*) as usage_count
        FROM returns
        WHERE user_email = ?
        AND status != 'REJECTED'
        GROUP BY LOWER(reason)
        HAVING usage_count > 2
    """
    result = db.execute(query, (user_email,)).fetchall()
    repeated_reasons = len(result)
    
    if repeated_reasons > 0:
        risk_score += 20
        risk_factors.append(f"Repeated reason patterns ({repeated_reasons} patterns)")
    
    # Determine risk level
    if risk_score >= 50:
        risk_reason = "HIGH risk: " + ", ".join(risk_factors)
    elif risk_score >= 25:
        risk_reason = "MEDIUM risk: " + ", ".join(risk_factors)
    else:
        risk_reason = "LOW risk: no suspicious patterns"
    
    check_result: FraudCheckResult = {
        "check_type": "reason",
        "score": min(100, risk_score),
        "reason": risk_reason,
        "details": {"reason_length": len(return_reason), "risk_factors": risk_factors}
    }
    
    return {"fraud_checks": [check_result]}


def check_timing_patterns(state: FraudDetectionState) -> dict:
    """Check 4: Timing pattern analysis (15% weight).
    
    Analyzes days between order and return:
    - Very quick returns (≤2 days) = suspicious
    - Quick returns (≤7 days) = medium risk
    - Near deadline (≥20 days) = gaming the system
    - Normal window (8-19 days) = legitimate usage
    """
    order_id = state["order_id"]
    
    db = get_db_connection()
    
    # Get order date
    query = """
        SELECT order_date, status
        FROM orders
        WHERE order_id = ?
    """
    result = db.execute(query, (order_id,)).fetchone()
    
    if not result:
        # Can't determine timing, assign neutral score
        check_result: FraudCheckResult = {
            "check_type": "timing",
            "score": 0,
            "reason": "Unable to determine order timing",
            "details": {"error": "Order not found"}
        }
        return {"fraud_checks": [check_result]}
    
    # Parse order date - handle both formats (with and without time)
    order_date_str = result['order_date']
    try:
        if len(order_date_str) > 10:  # Has time component
            order_date = datetime.strptime(order_date_str, '%Y-%m-%d %H:%M:%S')
        else:  # Date only
            order_date = datetime.strptime(order_date_str, '%Y-%m-%d')
    except ValueError:
        # Fallback - try both formats
        try:
            order_date = datetime.strptime(order_date_str, '%Y-%m-%d %H:%M:%S')
        except ValueError:
            order_date = datetime.strptime(order_date_str, '%Y-%m-%d')
    
    days_since_order = (datetime.now() - order_date).days
    
    # Calculate risk based on timing
    if days_since_order <= 2:
        risk_score = 85
        risk_reason = f"HIGH risk: Very quick return (within 2 days) ({days_since_order} days since order)"
    elif days_since_order <= 7:
        risk_score = 50
        risk_reason = f"MEDIUM risk: Quick return ({days_since_order} days since order)"
    elif days_since_order >= 20:
        risk_score = 35
        risk_reason = f"MEDIUM risk: Near deadline return ({days_since_order} days, gaming 30-day window?)"
    else:
        risk_score = 15
        risk_reason = f"LOW risk: Normal return timing ({days_since_order} days since order)"
    
    check_result: FraudCheckResult = {
        "check_type": "timing",
        "score": risk_score,
        "reason": risk_reason,
        "details": {"days_since_order": days_since_order, "order_date": order_date_str}
    }
    
    return {"fraud_checks": [check_result]}

def spawn_fraud_checks(state: FraudDetectionState) -> list[Send]:
    """Router function for dynamic parallel processing.
    
    Returns a list of Send objects - one for each fraud check.
    All checks execute simultaneously!
    """
    return [
        Send("check_frequency", state),
        Send("check_value", state),
        Send("check_reason", state),
        Send("check_timing", state)
    ]


# ==================== AGGREGATION ====================

def aggregate_fraud_scores(state: FraudDetectionState) -> dict:
    """Aggregate results from all fraud checks with weighted scoring.
    
    Weights:
    - Frequency: 35% (most important - serial returners)
    - Value: 35% (high-value fraud is costly)
    - Reason: 15% (supporting indicator)
    - Timing: 15% (supporting indicator)
    """
    fraud_checks = state["fraud_checks"]
    
    # Weight map
    weights = {
        "frequency": 0.35,
        "value": 0.35,
        "reason": 0.15,
        "timing": 0.15
    }
    
    # Calculate weighted score
    total_score = 0
    individual_checks = {}
    
    for check in fraud_checks:
        check_type = check["check_type"]
        score = check["score"]
        weight = weights.get(check_type, 0)
        
        total_score += score * weight
        individual_checks[check_type] = {
            "score": score,
            "reason": check["reason"],
            "details": check["details"],
            "weight": weight
        }
    
    # Determine risk level
    if total_score >= 60:
        risk_level = "HIGH"
        recommendation = "Admin review required - high fraud risk"
    elif total_score >= 30:
        risk_level = "MEDIUM"
        recommendation = "Admin review recommended"
    else:
        risk_level = "LOW"
        recommendation = "Safe to approve automatically"
    
    fraud_details = {
        "risk_level": risk_level,
        "recommendation": recommendation,
        "individual_checks": individual_checks,
        "total_weighted_score": total_score
    }
    
    return {
        "fraud_score": round(total_score, 2),
        "fraud_details": fraud_details
    }


# ==================== GRAPH BUILDER ====================

def create_fraud_detection_subgraph() -> StateGraph:
    """Build fraud detection subgraph with parallel processing."""
    
    graph = StateGraph(FraudDetectionState)
    
    # Add all check nodes
    graph.add_node("check_frequency", check_return_frequency)
    graph.add_node("check_value", check_order_value_patterns)
    graph.add_node("check_reason", check_return_reason_patterns)
    graph.add_node("check_timing", check_timing_patterns)
    graph.add_node("aggregate_fraud_scores", aggregate_fraud_scores)
    
    # CRITICAL: Use conditional_edges with Send function for parallel execution
    # spawn_fraud_checks is NOT added as a node - it's a router function
    graph.add_conditional_edges(
        START,
        spawn_fraud_checks,
        ["check_frequency", "check_value", "check_reason", "check_timing"]
    )
    
    # All checks converge to aggregator
    graph.add_edge("check_frequency", "aggregate_fraud_scores")
    graph.add_edge("check_value", "aggregate_fraud_scores")
    graph.add_edge("check_reason", "aggregate_fraud_scores")
    graph.add_edge("check_timing", "aggregate_fraud_scores")
    
    graph.add_edge("aggregate_fraud_scores", END)
    
    return graph.compile()
